  function switchTableView(state) {
    const tableEl = document.getElementById('state-table');
    const emptyEl = document.getElementById('state-empty');
    const skeletonEl = document.getElementById('state-skeleton');

    const btnTable = document.getElementById('btn-view-table');
    const btnEmpty = document.getElementById('btn-view-empty');
    const btnSkel = document.getElementById('btn-view-skeleton');

    [btnTable, btnEmpty, btnSkel].forEach(b => {
      b.className = 'px-3 py-1 rounded text-secondary hover:text-on-surface font-label-md text-label-md transition-all';
    });

    tableEl.classList.add('hidden');
    emptyEl.classList.add('hidden');
    skeletonEl.classList.add('hidden');

    if (state === 'table') {
      tableEl.classList.remove('hidden');
      btnTable.className = 'px-3 py-1 rounded text-on-primary bg-primary-container font-label-md text-label-md transition-all shadow-sm';
    } else if (state === 'empty') {
      emptyEl.classList.remove('hidden');
      btnEmpty.className = 'px-3 py-1 rounded text-on-primary bg-primary-container font-label-md text-label-md transition-all shadow-sm';
    } else if (state === 'skeleton') {
      skeletonEl.classList.remove('hidden');
      btnSkel.className = 'px-3 py-1 rounded text-on-primary bg-primary-container font-label-md text-label-md transition-all shadow-sm';
    }
  }
